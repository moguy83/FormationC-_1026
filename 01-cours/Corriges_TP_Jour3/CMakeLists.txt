cmake_minimum_required(VERSION 3.24)
project(CorrigesJour3 VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BUILD_TESTS "Compiler et enregistrer les tests" ON)
option(BUILD_DOCS "Generer la documentation Doxygen" OFF)
option(BUILD_INTEGRATION_TESTS "Tests PostgreSQL reels (libpq requis)" OFF)
option(BUILD_EXPECTED "Compiler TP10C avec std::expected (C++23)" ON)
option(ENABLE_COVERAGE "Instrumenter pour gcovr avec GCC" OFF)
option(ENABLE_SANITIZERS "Activer ASan et UBSan" OFF)
include(FetchContent)
# Réutiliser d'abord les paquets du système, sinon télécharger des versions fixées.
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3)
  FetchContent_MakeAvailable(json)
endif()
find_package(spdlog 1.12 QUIET)
if(NOT spdlog_FOUND)
  FetchContent_Declare(spdlog URL https://codeload.github.com/gabime/spdlog/tar.gz/refs/tags/v1.14.1)
  FetchContent_MakeAvailable(spdlog)
endif()
add_library(options_cours INTERFACE)
target_compile_features(options_cours INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(options_cours INTERFACE /W4)
else()
  target_compile_options(options_cours INTERFACE -Wall -Wextra -Wpedantic)
endif()
if(ENABLE_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Ce profil gcovr utilise GCC/gcov")
  endif()
  target_compile_options(options_cours INTERFACE --coverage -O0 -g)
  target_link_options(options_cours INTERFACE --coverage)
endif()
if(ENABLE_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Ce profil ASan+UBSan necessite GCC ou Clang")
  endif()
  target_compile_options(options_cours INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(options_cours INTERFACE -fsanitize=address,undefined)
endif()
if(BUILD_TESTS)
  enable_testing()
endif()
# Une cible par fichier contenant main(), jamais tous les .cpp dans un même exécutable.
function(exercice nom source)
  add_executable(${nom} ${source})
  target_link_libraries(${nom} PRIVATE options_cours)
  target_include_directories(${nom} PRIVATE ${PROJECT_SOURCE_DIR}/commun)
  if(BUILD_TESTS)
    file(MAKE_DIRECTORY ${PROJECT_BINARY_DIR}/travail/${nom})
    add_test(NAME ${nom} COMMAND ${nom})
    set_tests_properties(${nom} PROPERTIES WORKING_DIRECTORY ${PROJECT_BINARY_DIR}/travail/${nom} TIMEOUT 90)
  endif()
endfunction()
exercice(tp09a TP09_templates/tp09a_tableau.cpp)
exercice(tp09b TP09_templates/tp09b_algorithmes.cpp)
exercice(tp09c TP09_templates/tp09c_serialiseur.cpp)
target_link_libraries(tp09c PRIVATE nlohmann_json::nlohmann_json)
exercice(tp09d TP09_templates/tp09d_persistable.cpp)
exercice(tp10a TP10_exceptions/tp10a_calculatrice.cpp)
exercice(tp10b TP10_exceptions/tp10b_fichier_config.cpp)
if(BUILD_EXPECTED)
  include(CheckCXXSourceCompiles)
  set(CMAKE_CXX_STANDARD 23)
  check_cxx_source_compiles("#include <expected>
    int main(){std::expected<int,int> r=42; return *r==42?0:1;}" HAS_EXPECTED)
  unset(CMAKE_CXX_STANDARD)
  if(HAS_EXPECTED)
    exercice(tp10c TP10_exceptions/tp10c_strategies.cpp)
    target_compile_features(tp10c PRIVATE cxx_std_23)
  else()
    message(WARNING "TP10C non compile : std::expected absent. Utiliser GCC 13+ avec libstdc++ recente.")
  endif()
endif()
