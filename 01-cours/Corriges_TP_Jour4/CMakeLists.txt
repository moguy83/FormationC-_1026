cmake_minimum_required(VERSION 3.24)
project(CorrigesJour4 VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BUILD_TESTS "Enregistrer les tests CTest" ON)
option(BUILD_RACE_DEMO "Compiler la demonstration volontairement fautive" OFF)
option(ENABLE_ASAN "Activer AddressSanitizer + UBSan" OFF)
option(ENABLE_TSAN "Activer ThreadSanitizer" OFF)
if(ENABLE_ASAN AND ENABLE_TSAN)
  message(FATAL_ERROR "ASan et TSan doivent utiliser des builds separes")
endif()
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3)
  FetchContent_MakeAvailable(json)
endif()
add_library(options_cours INTERFACE)
target_include_directories(options_cours INTERFACE ${PROJECT_SOURCE_DIR}/commun)
if(MSVC)
  target_compile_options(options_cours INTERFACE /W4)
  if(ENABLE_ASAN OR ENABLE_TSAN)
    message(FATAL_ERROR "Les profils fournis de sanitizers utilisent GCC/Clang")
  endif()
else()
  target_compile_options(options_cours INTERFACE -Wall -Wextra -Wpedantic)
  if(ENABLE_ASAN)
    target_compile_options(options_cours INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(options_cours INTERFACE -fsanitize=address,undefined)
  endif()
  if(ENABLE_TSAN)
    target_compile_options(options_cours INTERFACE -fsanitize=thread -g)
    target_link_options(options_cours INTERFACE -fsanitize=thread)
  endif()
endif()
if(BUILD_TESTS)
  enable_testing()
endif()
function(exercice nom source)
  add_executable(${nom} ${source})
  target_compile_features(${nom} PRIVATE cxx_std_20)
  target_link_libraries(${nom} PRIVATE options_cours Threads::Threads)
  if(BUILD_TESTS)
    file(MAKE_DIRECTORY ${PROJECT_BINARY_DIR}/travail/${nom})
    add_test(NAME ${nom} COMMAND ${nom} ${ARGN})
    set_tests_properties(${nom} PROPERTIES WORKING_DIRECTORY ${PROJECT_BINARY_DIR}/travail/${nom} TIMEOUT 90)
  endif()
endfunction()
